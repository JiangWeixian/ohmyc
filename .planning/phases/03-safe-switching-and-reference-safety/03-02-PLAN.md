Preflight endpoint (GET /api/profiles/:name/preflight):
```typescript
interface PreflightResult {
  canActivate: boolean;
  missing: string[];           // e.g., ['agent:reviewer', 'skill:deploy']
  settingsWarnings: string[];  // e.g., ["Settings key 'model' would be overwritten"]
  currentActive: string | null;
}
```

Activate endpoint responses:
- 200: `{ success: true, warnings: string[] }`
- 422: `{ error: string, missing: string[] }` (ActivationBlockedError)
- 423: `{ error: string }` (Lock contention)

Delete endpoint response for active profile:
- 409: `{ error: "Cannot delete active profile. Deactivate <name> before deleting it." }`

       async function fetchPreflight(name: string): Promise<{
         canActivate: boolean;
         missing: string[];
         settingsWarnings: string[];
         currentActive: string | null;
       }> {
         const res = await fetch(`/api/profiles/${encodeURIComponent(name)}/preflight`);
         if (!res.ok) {
           const err = await res.json();
           throw new Error(err.error || 'Failed to run preflight check');
         }
         return res.json();
       }

       export function usePreflight() {
         return useMutation({
           mutationFn: fetchPreflight,
         });
       }
