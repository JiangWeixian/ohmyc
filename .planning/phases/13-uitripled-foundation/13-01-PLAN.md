1. **Install Radix UI peer dependencies** into packages/ui. Run from packages/ui directory:
   ```
   npm install @radix-ui/react-slot@^1.2.3 @radix-ui/react-dialog@^1.1.15 @radix-ui/react-tabs@^1.1.13 @radix-ui/react-select@^2.2.6 @radix-ui/react-switch@^1.2.6 @radix-ui/react-tooltip@^1.2.8 @radix-ui/react-dropdown-menu@^2.1.16 @radix-ui/react-separator@^1.1.8 @radix-ui/react-avatar@^1.1.11 @radix-ui/react-slider@^1.3.6 @radix-ui/react-checkbox@^1.3.3 class-variance-authority@^0.7.1
   ```
   These are the exact versions from vendor/uitripled/packages/components/react-shadcn/package.json dependencies. Do NOT install next, next-themes, recharts, @dnd-kit/*, or @uitripled/utils — those are not needed for primitives.

2. **Create packages/ui/tsconfig.json** with path aliases:
   ```json
   {
     "compilerOptions": {
       "target": "ES2020",
       "useDefineForClassFields": true,
       "lib": ["ES2020", "DOM", "DOM.Iterable"],
       "module": "ESNext",
       "skipLibCheck": true,
       "moduleResolution": "bundler",
       "allowImportingTsExtensions": true,
       "resolveJsonModule": true,
       "isolatedModules": true,
       "noEmit": true,
       "jsx": "react-jsx",
       "strict": true,
       "noUnusedLocals": true,
       "noUnusedParameters": true,
       "noFallthroughCasesInSwitch": true,
       "baseUrl": ".",
       "paths": {
         "@/*": ["./src/*"]
       }
     },
     "include": ["src"]
   }
   ```
   The critical setting is `paths: { "@/*": ["./src/*"] }` so uitripled's `import { cn } from "@/lib/utils"` resolves correctly.

3. **Update packages/ui/vite.config.ts** to add the resolve alias:
   ```typescript
   import { defineConfig } from 'vite'
   import react from '@vitejs/plugin-react'
   import path from 'path'

   export default defineConfig({
     plugins: [react()],
     resolve: {
       alias: {
         '@': path.resolve(__dirname, './src'),
       },
     },
     server: {
       proxy: {
         '/api': 'http://127.0.0.1:3000',
       },
     },
   })
   ```

4. **Create packages/ui/src/lib/utils.ts** — this is the @/lib/utils path that uitripled primitives import from. The content is identical to the existing cn.ts:
   ```typescript
   import { clsx, type ClassValue } from "clsx";
   import { twMerge } from "tailwind-merge";

   export function cn(...inputs: ClassValue[]) {
     return twMerge(clsx(inputs));
   }
   ```
   Note: Keep the existing packages/ui/src/components/cn.ts in place (existing components import from it). The new file at @/lib/utils serves the uitripled primitives. Both export the same function.
  </action>
