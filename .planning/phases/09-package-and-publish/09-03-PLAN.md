```typescript
import { defineConfig } from 'tsup';
import { cpSync, existsSync } from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

export default defineConfig({
  entry: ['src/index.ts'],
  format: ['cjs'],
  splitting: false,
  clean: true,
  noExternal: [
    '@claudeui/shared',
    '@fastify/static',
    'cac',
    'fastify',
    'get-port',
    'gray-matter',

    const uiDist = path.resolve(__dirname, '../ui/dist');
    const cliUiDist = path.resolve(__dirname, 'dist/ui');

    if (existsSync(uiDist)) {
      cpSync(uiDist, cliUiDist, { recursive: true });
      console.log(`Copied UI assets from ${uiDist} to ${cliUiDist}`);
    } else {
      console.warn(`Warning: UI dist not found at ${uiDist}. Run 'pnpm --filter @claudeui/ui build' first.`);
    }
  },
});
```

Why CJS instead of ESM: The bundle has 84 `__require()` calls from CJS dependencies (fastify, avvio, etc.) that call `require('events')`, `require('fs')`, etc. In ESM format, tsup creates a shim that throws because `require` is undefined. In CJS format, `require` exists natively, so all these calls work. The CLI entry has a `#!/usr/bin/env node` shebang and runs under Node.js -- CJS is the native format. The package already has `"type": "module"` in package.json, but the bundled dist/index.js with CJS format will be run directly by Node via the bin entry, so the `type` field does not interfere.

Note: Do NOT add `bundle: true` or change the `onSuccess` hook. Only change `format: ['esm']` to `format: ['cjs']`.

```javascript
distTestSkip('dist/index.js has no external npm package imports', () => {
  const bundleContent = readFileSync(distIndexPath, 'utf-8');

  // Check ESM from imports
  const esmImports = bundleContent.match(
    /(?<=from\s['"])(?!node:|\.\/|\.\.\/)[^'"]+(?=['"])/g,
  ) || [];

  // Check CJS require calls for npm packages
  // Match require("X") or require('X') but not __require("X") (internal shim)
  const cjsImports = bundleContent.match(
    /(?<![_\w])require\(\s*['"]([^'"]+)['"]\s*\)/g,
  ) || [];
  // Extract just the module names from require() calls
  const cjsSpecifiers = cjsImports.map(m => {
    const match = m.match(/require\(\s*['"]([^'"]+)['"]\s*\)/);
    return match ? match[1] : '';
  });

  // Filter out Node.js built-in modules
  const npmImports = [...esmImports, ...cjsSpecifiers]
    .filter((spec) => !NODE_BUILTINS.has(spec));

  expect(npmImports).toEqual([]);
});
```
